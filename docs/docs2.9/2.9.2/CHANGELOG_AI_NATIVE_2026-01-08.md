## 🔧 Technical Details

### Database Schema

**ai_compute.db:**
```sql
-- AI Tasks
CREATE TABLE ai_tasks (
    task_id TEXT PRIMARY KEY,
    task_type TEXT,
    created_at REAL,
    deadline REAL,
    reward REAL,
    model TEXT,
    input_data TEXT,
    status TEXT,
    assigned_miner TEXT,
    result TEXT
);

-- Miners
CREATE TABLE miners (
    miner_address TEXT PRIMARY KEY,
    consciousness_level INTEGER,
    hashrate REAL,
    gpu_available INTEGER,
    tasks_completed INTEGER,
    reputation REAL
);

-- Task Results
CREATE TABLE task_results (
    task_id TEXT,
    miner_address TEXT,
    completed_at REAL,
    execution_time REAL,
    reward_paid REAL,
    quality_score REAL
);
```

**zion_ai_memory.db:**
```sql
-- Git Commits
CREATE TABLE git_commits (
    hash TEXT PRIMARY KEY,
    author TEXT,
    date TEXT,
    message TEXT
);

-- Session Reports
CREATE TABLE sessions (
    id INTEGER PRIMARY KEY,
    filename TEXT,
    date TEXT,
    title TEXT,
    content TEXT
);

-- Code Snippets
CREATE TABLE code_snippets (
    id INTEGER PRIMARY KEY,
    session_file TEXT,
    language TEXT,
    code TEXT,
    context TEXT
);

-- Full-text Search
CREATE VIRTUAL TABLE memory_fts USING fts5(
    source,
    date,
    content
);
```
